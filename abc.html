import { Component } from '@angular/core'; import { ICellRendererAngularComp } from 'ag-grid-angular'; import { ICellRendererParams } from 'ag-grid-community'; import { Candidate } from '../../../core/models/candidate.model'; import { RoundStatus } from '../../../core/models/round-status.enum'; import { WorkflowVisibilityService } from '../../../core/services/workflow-visibility.service'; @Component({ selector: 'app-workflow-pipeline-cell', template: `{{ label }}`, styleUrls: ['./workflow-pipeline-cell.component.scss'] }) export class WorkflowPipelineCellComponent implements ICellRendererAngularComp { label = ''; constructor(private visibility: WorkflowVisibilityService) {} agInit(params: ICellRendererParams): void { this.label = this.deriveLabel(params.data); } refresh(): boolean { return false; } private deriveLabel(c: Candidate): string { if (c.hrRound.selection === RoundStatus.Selected) return 'Offer Stage'; if (c.onShoreRound.selection === RoundStatus.Selected) return 'HR Round'; if (c.mgmtRound.selection === RoundStatus.Selected) return 'On-Shore Round'; if (this.visibility.techStageDecision(c) === RoundStatus.Selected) return 'Management Round'; if (c.techRound1.selection === RoundStatus.Selected) return 'Tech Round 2'; if (c.prescreeningSelected === RoundStatus.Selected) return 'Tech Round 1'; if (this.visibility.techStageDecision(c) === RoundStatus.Rejected || c.mgmtRound.selection === RoundStatus.Rejected || c.onShoreRound.selection === RoundStatus.Rejected) return 'Rejected'; return 'Pre-Screen'; } }
