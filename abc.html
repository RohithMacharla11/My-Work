/** * mapToOnboardingTemplate.js * * Reads the single .csv file found in ./data/, remaps its columns onto the * Onboarding Template column layout, and writes the result to * ./output/Onboarding_Template_ .xlsx>.xlsx * * Usage: * npm install exceljs * node mapToOnboardingTemplate.js */ const fs = require("fs"); const path = require("path"); const ExcelJS = require("exceljs"); // --------------------------------------------------------------------------- // Minimal CSV parser (handles quoted fields, embedded commas/newlines, "") // --------------------------------------------------------------------------- function parseCsv(text) { const rows = []; let row = []; let field = ""; let inQuotes = false; // Normalize line endings so \r\n inside quotes doesn't break things unexpectedly for (let i = 0; i < text.length; i++) { const ch = text[i]; const next = text[i + 1]; if (inQuotes) { if (ch === '"' && next === '"') { field += '"'; i++; } else if (ch === '"') { inQuotes = false; } else { field += ch; } } else { if (ch === '"') { inQuotes = true; } else if (ch === ",") { row.push(field); field = ""; } else if (ch === "\r") { // skip, handled by \n } else if (ch === "\n") { row.push(field); rows.push(row); row = []; field = ""; } else { field += ch; } } } // last field/row (file may not end with newline) if (field.length > 0 || row.length > 0) { row.push(field); rows.push(row); } // Drop fully-empty trailing rows while (rows.length > 0 && rows[rows.length - 1].every((c) => c === "")) { rows.pop(); } return rows; } // --------------------------------------------------------------------------- // Configuration // --------------------------------------------------------------------------- const DATA_DIR = "data"; const OUTPUT_DIR = "output"; // Template column order (exact headers to write into the output file) const TEMPLATE_COLUMNS = [ "Record Type", "Country", "Application Country", "Visitors", "Report Code", "Pattern Search", "Filename Pattern RegEx", "Filename Pattern", "Record Description", "Office of Record", "Retention Requirement", "Retention period", "Regulatory Body", "Regulatory Body State", "Regulatory Body Country", "Regulatory Rule", "Data Classification", "IT Storage Requirement", "Primary Business Contact", "Primary IT Contact", "Disposition After", "Disposition", "Pattern Match Order", "Owners", "Limited Visitors", "LegalHold Visitors", "Manual Upload Members", "Members", "Application Owner", "Application Support Group", "Region", "IsDisposalDynamic", "IsDisposal", "Primary Business Contact Email", "Primary IT Contact Email", ]; // Columns that are OPTIONAL: if the source cell is blank, leave the template // cell blank too (do NOT fill with "NA"). const OPTIONAL_BLANK_COLUMNS = new Set([ "Country", "Application Country", "Manual Upload Members", "Members", ]); // Every template column not in OPTIONAL_BLANK_COLUMNS is MANDATORY: // a blank source cell becomes "NA" in the output. // Columns where a blank source cell should become 0 (not "NA"). // TRUE/FALSE in these columns are still converted to 1/0 as usual. const ZERO_DEFAULT_COLUMNS = new Set([ "Pattern Search", "IsDisposal", "IsDisposalDynamic", "Pattern Match Order", ]); // Columns that should be stored as real numbers in the output (not text), // whenever the value is numeric. const NUMERIC_COLUMNS = new Set([ "Pattern Search", "IsDisposal", "IsDisposalDynamic", "Pattern Match Order", ]); // When a single cell holds more than one person, names are comma-separated // and their emails are semicolon-separated, in the same order. const NAME_DELIMITER = ","; const EMAIL_DELIMITER = ";"; // Contact/email pairs: if a row has a contact name but a blank email, and // another row has the same contact name WITH an email, fill it in. const CONTACT_EMAIL_PAIRS = [ { contactCol: "Primary Business Contact", emailCol: "Primary Business Contact Email" }, { contactCol: "Primary IT Contact", emailCol: "Primary IT Contact Email" }, ]; // --------------------------------------------------------------------------- // Helpers // --------------------------------------------------------------------------- function normalizeHeader(name) { return String(name ?? "") .toLowerCase() .replace(/[^a-z0-9]/g, ""); } function isBlank(value) { if (value === null || value === undefined) return true; if (typeof value === "string" && value.trim() === "") return true; return false; } function isBlankCellValue(value) { return value === "NA" || value === "" || value === undefined || value === null; } function splitList(value, delimiter) { if (isBlankCellValue(value)) return []; return String(value) .split(delimiter) .map((s) => s.trim()) .filter((s) => s.length > 0); } function normalizeValue(value, { optional = false, zeroDefault = false, numeric = false } = {}) { if (isBlank(value)) { if (zeroDefault) return 0; return optional ? "" : "NA"; } if (typeof value === "boolean") { return value ? 1 : 0; } if (typeof value === "string") { const v = value.trim(); if (v.toUpperCase() === "TRUE") return 1; if (v.toUpperCase() === "FALSE") return 0; if (numeric && /^-?\d+(\.\d+)?$/.test(v)) return Number(v); return v; } // ExcelJS may hand back rich objects (formulas, hyperlinks, dates). // Fall back to their text/result if present, else the raw value. if (typeof value === "object") { if ("result" in value) return normalizeValue(value.result, { optional, zeroDefault, numeric }); if ("text" in value) return normalizeValue(value.text, { optional, zeroDefault, numeric }); if (value instanceof Date) return value; } return value; } function findSingleInputFile() { if (!fs.existsSync(DATA_DIR)) { console.error(`No '${DATA_DIR}/' folder found. Create it and place one .csv file inside.`); process.exit(1); } const candidates = fs .readdirSync(DATA_DIR) .filter((f) => f.toLowerCase().endsWith(".csv") && !f.startsWith("~$")); if (candidates.length === 0) { console.error(`No .csv file found in '${DATA_DIR}/'. Place exactly one file there.`); process.exit(1); } if (candidates.length > 1) { console.error( `Expected exactly one .csv file in '${DATA_DIR}/', found ${candidates.length}: ${candidates.join(", ")}` ); process.exit(1); } return path.join(DATA_DIR, candidates[0]); } // --------------------------------------------------------------------------- // Main // --------------------------------------------------------------------------- async function main() { const inputPath = findSingleInputFile(); const inputFilename = path.basename(inputPath); console.log(`Reading source file: ${inputPath}`); const rawText = fs.readFileSync(inputPath, "utf8").replace(/^\uFEFF/, ""); // strip BOM if present const csvRows = parseCsv(rawText); if (csvRows.length === 0) { console.error("The CSV file appears to be empty."); process.exit(1); } const sourceHeaders = csvRows[0]; const dataRows = csvRows.slice(1); const dataRowCount = dataRows.length; // Map normalized header -> column index (0-based) const sourceHeaderByNormalized = {}; sourceHeaders.forEach((h, idx) => { const norm = normalizeHeader(h); if (norm) sourceHeaderByNormalized[norm] = idx; }); // Build output data: array of column-arrays, in TEMPLATE_COLUMNS order const unmatched = []; const columnsData = []; // columnsData[colIdx] = array of values for each data row for (const templateCol of TEMPLATE_COLUMNS) { const norm = normalizeHeader(templateCol); const optional = OPTIONAL_BLANK_COLUMNS.has(templateCol); const zeroDefault = ZERO_DEFAULT_COLUMNS.has(templateCol); const numeric = NUMERIC_COLUMNS.has(templateCol); const sourceColIdx = sourceHeaderByNormalized[norm]; const values = []; if (sourceColIdx !== undefined) { for (let r = 0; r < dataRowCount; r++) { const rawValue = dataRows[r][sourceColIdx]; values.push(normalizeValue(rawValue, { optional, zeroDefault, numeric })); } } else { unmatched.push(templateCol); const fill = zeroDefault ? 0 : optional ? "" : "NA"; for (let r = 0; r < dataRowCount; r++) values.push(fill); } columnsData.push(values); } // --------------------------------------------------------------------- // Fill down contact emails: if the same contact name appears on another // row with a non-blank email, use that email for rows where it's missing. // --------------------------------------------------------------------- const colIndex = {}; TEMPLATE_COLUMNS.forEach((col, idx) => (colIndex[col] = idx)); for (const { contactCol, emailCol } of CONTACT_EMAIL_PAIRS) { const contactIdx = colIndex[contactCol]; const emailIdx = colIndex[emailCol]; if (contactIdx === undefined || emailIdx === undefined) continue; const contactValues = columnsData[contactIdx]; const emailValues = columnsData[emailIdx]; // Build name -> known email map, tracking ambiguity. // Cells can hold multiple people: names comma-separated, emails // semicolon-separated, matched by position. Rows where the two lists // don't line up 1:1 are skipped for map-building (can't pair reliably). const nameToEmail = {}; const ambiguousNames = new Set(); for (let r = 0; r < dataRowCount; r++) { const names = splitList(contactValues[r], NAME_DELIMITER); const emails = splitList(emailValues[r], EMAIL_DELIMITER); if (names.length === 0 || emails.length === 0) continue; if (names.length !== emails.length) continue; // can't pair reliably for (let i = 0; i < names.length; i++) { const key = names[i].toLowerCase(); const email = emails[i]; if (!(key in nameToEmail)) { nameToEmail[key] = email; } else if (nameToEmail[key] !== email) { ambiguousNames.add(key); } } } // Fill blanks: only when EVERY name in the row has exactly one known, // unambiguous email. Otherwise leave the cell as-is — no guessing. for (let r = 0; r < dataRowCount; r++) { if (!isBlankCellValue(emailValues[r])) continue; // already has a value const names = splitList(contactValues[r], NAME_DELIMITER); if (names.length === 0) continue; // no contact name to look up const resolvedEmails = []; let allKnown = true; for (const name of names) { const key = name.toLowerCase(); if (key in nameToEmail && !ambiguousNames.has(key)) { resolvedEmails.push(nameToEmail[key]); } else { allKnown = false; break; } } if (allKnown) { emailValues[r] = resolvedEmails.join(`${EMAIL_DELIMITER} `); } // else: leave as "NA" — not every name has a confident match. } } if (unmatched.length > 0) { console.log("NOTE: no matching source column found for:", unmatched); console.log(" (filled per the blank/NA rule for those columns)"); } // Write output workbook const outWorkbook = new ExcelJS.Workbook(); const outSheet = outWorkbook.addWorksheet("Metadata"); outSheet.addRow(TEMPLATE_COLUMNS); outSheet.getRow(1).eachCell((cell) => { cell.font = { name: "Arial", bold: true }; }); for (let r = 0; r < dataRowCount; r++) { const rowValues = columnsData.map((col) => col[r]); const row = outSheet.addRow(rowValues); row.eachCell({ includeEmpty: true }, (cell) => { cell.font = { name: "Arial" }; }); } // Reasonable default column width outSheet.columns.forEach((col) => { col.width = 22; }); if (!fs.existsSync(OUTPUT_DIR)) fs.mkdirSync(OUTPUT_DIR, { recursive: true }); const inputBaseName = path.basename(inputFilename, path.extname(inputFilename)); const outputFilename = `Onboarding_Template_${inputBaseName}.xlsx`; const outputPath = path.join(OUTPUT_DIR, outputFilename); await outWorkbook.xlsx.writeFile(outputPath); console.log(`Done. Wrote ${dataRowCount} rows to: ${outputPath}`); } main().catch((err) => { console.error("Error:", err); process.exit(1); });
