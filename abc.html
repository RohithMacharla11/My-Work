async onSave(action: 'draft' | 'submit' | 'approve' | 'throwback' | 'sendback' | 'reject' | 'adminSave'): Promise { console.log("onSave called"); console.log(this.permissions); if(!this.permissions?.canAdminSave && this.permissions?.isReadOnly) { this.showToast("You do not have permissions to edit this form",'error'); return; } // Flow C — Throwback/Sendback: always open mandatory comment modal // regardless of whether pendingComment has text or not if((action === 'throwback' || action === 'sendback') && !this.commentSaved) { this.pendingAction = action; this.pendingComment = ''; // always fresh, clear whatever was there this.openAddCommentModal(true); return; } if(action === 'submit' && !this.validateBeforeSubmit()) return; if(action === 'draft') this.draftSaved = true; this.setLoading(true); this.isSubmitting = true; this.submitAction = action; try { // Get digest console.log('Step:Authenticating with SharePoint...'); const digest = this.source === 'archive' ? await this.spService.getArchivalRequestDigest().toPromise() : await this.spService.getRequestDigest().toPromise(); // Build payload const formValues = this.recordForm.getRawValue(); const currentStatus = this.LoadedRecord?.Status ?? 'Pending with Sender Maker'; const nextStatus = action === 'adminSave' ? currentStatus : getNextStatus(currentStatus, action); const role = STATUS_TO_ROLE[this.LoadedRecord?.Status] || 'SenderMaker'; let column = ''; let mergedUsers: number[] = []; if(action !== 'adminSave') { switch(role){ case 'SenderMaker' : column = 'SenderMakers'; break; case 'SenderChecker' : column = 'SenderCheckers'; break; case 'ReceiverMaker' : column = 'ReceiverMakers'; break; case 'ReceiverChecker': column = 'ReceiverCheckers'; break; } const existingUsers: number[] = this.LoadedRecord?.[`${column}Id`]?.results?.map((u:any)=> u.Id || u ) ?? []; mergedUsers = Array.from(new Set([...existingUsers, this.userService.userId])); } const actionMsg: string = this.getAuditAction(action); const nextStatusForAudit = nextStatus || this.LoadedRecord?.Status || 'NA'; const uploadedNames = this.selectedFiles.map(f => `Uploaded: ${f.name}`); const deletedNames = this.existingFiles.filter(f => f.markedForDeletion && !f.isDeleted) .map(f=> `Deleted: ${f.name}`); const allChanges = [...uploadedNames, ...deletedNames]; const attachmentChanges = allChanges.length > 0 ? allChanges.join(',') : 'NA'; // Build comment based on action const comment = (() => { if(action === 'sendback') return this.recordForm.get('SendBackComments')?.value.trim() || 'NA'; if(action === 'throwback') return this.recordForm.get('ThrowbackComments')?.value.trim() || 'NA'; if((action === 'submit' || action === 'draft') && this.isSenderMaker) return this.recordForm.get('SenderRemarks')?.value.trim() || 'NA'; if((action === 'submit') && this.isReceiverMaker) return this.recordForm.get('ReceiverRemarks')?.value.trim() || 'NA'; return 'NA'; })(); // Build comment entry for View Comments // Use pendingComment (from modal) as the comment for throwback/sendback const currentComment = this.pendingComment?.trim() || null; const commentEntry = (this.commentSaved && this.pendingComment.trim()) ? (() => { const effectiveRole = this.resolveEffectiveRole(); const actionTag = this.getCommentActionTag(action); const addedBy = this.userService.uid || 'Unknown'; const addedOn = new Date().toLocaleString('en-GB', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit', hour12: false }); return `${effectiveRole};${actionTag};${addedBy};${addedOn};${this.pendingComment.trim()}`; })() : null; const existingComments = this.LoadedRecord?.Comments || ''; const newComments = commentEntry ? (existingComments ? `${commentEntry}||${existingComments}` : commentEntry) : existingComments; const now = new Date().toISOString(); const existingRecord = this.LoadedRecord; const effectiveRole = this.resolveEffectiveRole(); const actionDates: any = {}; if(effectiveRole === 'SenderMaker' && action === 'submit'){ actionDates.SenderMakerSubmit = now; } if(effectiveRole === 'SenderChecker' && action === 'approve') { if(!existingRecord?.SenderCheckerApproveInitial) { actionDates.SenderCheckerApproveInitial = now; } actionDates.SenderCheckerApproveLatest = now; } if(effectiveRole === 'SenderChecker') { if(action === 'reject') actionDates.SenderCheckerReject = now; if(action === 'sendback') actionDates.SenderCheckerSendback = now; } if(effectiveRole === 'ReceiverMaker') { if(action === 'throwback') actionDates.ReceiverMakerThrowback = now; if(action === 'submit') actionDates.ReceiverMakerSubmit = now; } if(effectiveRole === 'ReceiverChecker') { if(action === 'approve') actionDates.ReceiverCheckerApprove = now; if(action === 'sendback') actionDates.ReceiverCheckerSendback = now; } const itemPayload: any = { SenderId: this.userService.userId, Product: formValues.Product, Source: formValues.Source, BusinessType: formValues.BusinessType, Counter: formValues.Counter, TransactionCCY: formValues.TransactionCCY || null, SourceNumber: formValues.SourceNumber || null, CustomerName: formValues.CustomerName || null, UrgentFlag: formValues.UrgentFlag ? 'Yes' : 'No', BRRefNo: formValues.BRRefNo || null, ReceiverReference: formValues.ReceiverReference || null, CustomerAccount: formValues.CustomerAccount || null, TransactionAmount: formValues.TransactionAmount ? Number(formValues.TransactionAmount) : null, ExecutionDate: formValues.ExecutionDate ? new Date(formValues.ExecutionDate).toISOString() : null, SenderRemarks: effectiveRole === 'SenderMaker' && action === 'submit' ? currentComment : this.LoadedRecord?.SenderRemarks, ReceiverRemarks: effectiveRole === 'ReceiverMaker' && action === 'submit' ? currentComment : this.LoadedRecord?.ReceiverRemarks, ThrowbackComments: action === 'throwback' ? currentComment : this.LoadedRecord?.ThrowbackComments, SendBackComments: action === 'sendback' ? currentComment : this.LoadedRecord?.SendBackComments, Comments: newComments, Status: nextStatus, ...(action !== 'adminSave' && column ? { [`${column}Id`]: { results: mergedUsers } } : {}), ...(action === 'adminSave' && column ? { [`${column}Id`]: { results: mergedUsers } } : {}), AssignedToId: action === 'draft' ? this.userService.userId : action === 'adminSave' ? this.assignToId ?? null : null, AuditLog: this.buildAuditEntry(actionMsg, nextStatusForAudit, attachmentChanges, currentComment ?? 'NA'), ...actionDates, ...(this.source !== 'archive' ? { IsSendBack: action === 'sendback' ? 'Yes' : (this.LoadedRecord?.IsSendBack || 'No'), IsThrowBack: action === 'throwback' ? 'Yes' : (this.LoadedRecord?.IsThrowBack || 'No'), } : {}), }; console.log("Item payload", itemPayload); let itemId: number; let workflowId: string = ''; if(!this.isNewRecord){ // UPDATE existing record console.log('Step:Updating record...'); if(this.source === 'archive') { await this.spService.updateArchivalListItem(digest, this.editItemId, itemPayload).toPromise(); } else { await this.spService.updateListItem(digest, this.editItemId, itemPayload).toPromise(); } itemId = this.editItemId; workflowId = this.LoadedRecord?.NovusWorkFlowID; console.log('Success:Record Updated'); } else { // CREATE new record console.log('Step:Creating record in SharePoint...'); const createdItem = await this.spService.createListItem(digest, itemPayload).toPromise(); this.editItemId = createdItem.Id; itemId = this.editItemId; console.log('Success: Record ID Created,', itemId); console.log('Step:Generating Workflow ID...'); workflowId = await this.spService.generateWorkFlowId(this.editItemId); await this.spService.updateListItem(digest, this.editItemId, { NovusWorkFlowID: workflowId, Status: nextStatus, [`${column}Id`]: { results: [this.userService.userId] }, AuditLog: this.buildAuditEntry(actionMsg, nextStatusForAudit, attachmentChanges, currentComment ?? 'NA') }).toPromise(); console.log(`Success:Saving Workflow ID: ${workflowId}`); console.log('Step:Creating document folder...'); await this.spService.createFolder(digest, this.editItemId.toString()).toPromise(); console.log("Success:Folder Created"); } itemId = this.editItemId; if(!this.isNewRecord) workflowId = this.LoadedRecord?.NovusWorkFlowID; // DELETE marked files first const filesToDelete = this.existingFiles.filter(f => f.markedForDeletion && !f.isDeleted); for(let i = 0; i < filesToDelete.length; i++) { console.log(`Step:Deleting file: ${filesToDelete[i].name}`); const deletedName = `deleted_${filesToDelete[i].name}`; await this.spService.updateFileItem( digest, filesToDelete[i].fileItemId!, { isDeleted: true, FileLeafRef: deletedName } ).toPromise(); } // Delay between delete and upload if(filesToDelete.length > 0) { await new Promise(resolve => setTimeout(resolve, 2000)); } // UPLOAD new files for(let i = 0; i < this.selectedFiles.length; i++) { const f = this.selectedFiles[i]; console.log(`Uploading file ${(i+1)} of ${this.selectedFiles.length}: ${f.name}`); this.source === 'archive' ? await this.spService.uploadFileToArchive(digest, this.liveid.toString(), f).toPromise() : await this.spService.uploadFile(digest, itemId.toString(), f).toPromise(); } this.workflowId = workflowId; console.log("All steps complete"); const successMsgs: Record = { draft: `Saved Successfully. Workflow ID : ${workflowId}`, submit: `Submitted Successfully. Workflow ID : ${workflowId}`, approve: `Approved Successfully. Workflow ID : ${workflowId}`, throwback: `Returned Successfully. Workflow ID : ${workflowId}`, sendback: `Sent Back Successfully. Workflow ID : ${workflowId}`, reject: `Rejected Workflow ID: ${workflowId}`, adminSave: `Saved Successfully. Workflow ID : ${workflowId}`, }; this.showToast(successMsgs[action] ?? 'Done.', 'success'); if(this.source === 'archive') { this.LoadedRecord = await this.spService.getArchivalListItem(this.editItemId).toPromise(); } else { this.LoadedRecord = await this.spService.getListItem(this.editItemId).toPromise(); } await this.triggerEmailAfterSave(action, currentStatus); this.assignmentChanged = false; // Reset comment state always after onSave this.pendingComment = ''; this.commentSaved = false; this.pendingAction = ''; if(action === 'draft') { this.isNewRecord = false; this.selectedFiles = []; this.loadRecord(this.editItemId); } else { setTimeout(() => { this.goToSourcePage(); }, 3000); } } catch(err: any) { console.error('Submit error:', err); this.submitProgress = ''; this.showToast( `Error: ${err?.error?.message ?? err?.message ?? 'Unknown error'}. Please try again or contact your administrator.`, 'error' ); } finally { this.isSubmitting = false; this.setLoading(false); } }
